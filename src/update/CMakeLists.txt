add_library(bending_update STATIC UpdateClient.cpp PortablePackage.cpp
    include/bending/update/UpdateClient.h)
add_library(Bending::Update ALIAS bending_update)
set_target_properties(bending_update PROPERTIES AUTOMOC ON)
target_include_directories(bending_update PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(bending_update PUBLIC Qt6::Core PRIVATE Qt6::Network Bending::Application bending_options)
qt_add_resources(bending_update update_assets PREFIX "/update" FILES data/extract-package.ps1)

add_executable(bending_portable_launcher PortableLauncher.cpp)
target_link_libraries(bending_portable_launcher PRIVATE bending_options)
set_target_properties(bending_portable_launcher PROPERTIES OUTPUT_NAME "3d-bending-launcher")
if(MSVC)
    set_property(TARGET bending_portable_launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

    enable_language(RC)
    configure_file(PortableLauncher.rc.in "${PROJECT_BINARY_DIR}/generated/PortableLauncher.rc" @ONLY)
    target_sources(bending_portable_launcher PRIVATE "${PROJECT_BINARY_DIR}/generated/PortableLauncher.rc")
    target_link_libraries(bending_portable_launcher PRIVATE shell32)
    set_target_properties(bending_portable_launcher PROPERTIES WIN32_EXECUTABLE TRUE)
