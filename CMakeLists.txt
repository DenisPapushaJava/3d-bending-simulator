cmake_minimum_required(VERSION 3.24)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" BENDING_VERSION LIMIT_COUNT 1)
if(NOT BENDING_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-(alpha|beta|rc)\\.[1-9][0-9]*)?$")
    message(FATAL_ERROR "Invalid VERSION: ${BENDING_VERSION}")
endif()
set(BENDING_NUMERIC_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
project(BendingSimulator VERSION "${BENDING_NUMERIC_VERSION}" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)

option(BENDING_BUILD_APP "Build the Qt Widgets application" ON)
option(BENDING_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)

add_library(bending_options INTERFACE)
if(MSVC)
    target_compile_options(bending_options INTERFACE /W4 /permissive- /utf-8)
    if(BENDING_WARNINGS_AS_ERRORS)
        target_compile_options(bending_options INTERFACE /WX)
    endif()
else()
    target_compile_options(bending_options INTERFACE -Wall -Wextra -Wpedantic)
    if(BENDING_WARNINGS_AS_ERRORS)
        target_compile_options(bending_options INTERFACE -Werror)
    endif()
endif()

configure_file(cmake/Version.h.in generated/bending/Version.h @ONLY)
add_library(bending_core src/core/BendSequence.cpp)
target_include_directories(bending_core PUBLIC src "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(bending_core PRIVATE bending_options)

find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
# Older OCCT packages declare ExactVersion compatibility, so inspect the detected version.
set(BENDING_OCCT_VERSION "${OpenCASCADE_MAJOR_VERSION}.${OpenCASCADE_MINOR_VERSION}.${OpenCASCADE_MAINTENANCE_VERSION}")
if(BENDING_OCCT_VERSION VERSION_LESS 7.6.0 OR NOT BENDING_OCCT_VERSION VERSION_LESS 8.0.0)
    message(FATAL_ERROR "Use OCCT 7.6–7.9; found ${BENDING_OCCT_VERSION}")
endif()
add_library(bending_cad src/cad/Kernel.cpp)
target_include_directories(bending_cad PUBLIC src)
target_include_directories(bending_cad SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(bending_cad PRIVATE bending_options TKernel)

if(BENDING_BUILD_APP)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
    add_library(bending_viewer src/viewer/ViewerPlaceholder.cpp)
    target_include_directories(bending_viewer PUBLIC src)
    target_link_libraries(bending_viewer PUBLIC Qt6::Widgets PRIVATE bending_options)

    qt_add_executable(bending_simulator src/app/main.cpp src/ui/MainWindow.cpp)
    set_target_properties(bending_simulator PROPERTIES OUTPUT_NAME "3d-bending-simulator")
    if(WIN32)
        set_target_properties(bending_simulator PROPERTIES WIN32_EXECUTABLE TRUE)
    endif()
    target_link_libraries(bending_simulator PRIVATE
        bending_core bending_cad bending_viewer bending_options Qt6::Widgets)
    install(TARGETS bending_simulator RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
    install(FILES VERSION THIRD_PARTY_NOTICES.md DESTINATION "${CMAKE_INSTALL_DOCDIR}")
    if(EXISTS "${CMAKE_SOURCE_DIR}/LICENSE")
        install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DOCDIR}")
    endif()
    include(cmake/Packaging.cmake)
endif()

if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE bending_core bending_options)
    add_test(NAME core.sequence COMMAND core_tests)

    add_executable(cad_smoke tests/cad_smoke.cpp)
    target_include_directories(cad_smoke SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
    target_link_libraries(cad_smoke PRIVATE bending_cad bending_options TKPrim TKBRep TKTopAlgo)
    add_test(NAME cad.kernel COMMAND cad_smoke)
    if(BENDING_BUILD_APP)
        add_test(NAME app.startup COMMAND bending_simulator --smoke-test)
        set_tests_properties(app.startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
    endif()
endif()
