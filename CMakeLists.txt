cmake_minimum_required(VERSION 3.24)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" BENDING_VERSION LIMIT_COUNT 1)
if(NOT BENDING_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-(alpha|beta|rc)\\.[1-9][0-9]*)?$")
    message(FATAL_ERROR "Invalid VERSION: ${BENDING_VERSION}")
endif()
set(BENDING_NUMERIC_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
project(BendingSimulator VERSION "${BENDING_NUMERIC_VERSION}" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)

option(BENDING_BUILD_APP "Build the Qt Widgets application" ON)
option(BENDING_BUILD_CAD "Build the OCCT adapter and its integration test" ON)
option(BENDING_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)

if(BENDING_BUILD_APP AND NOT BENDING_BUILD_CAD)
    message(FATAL_ERROR "BENDING_BUILD_APP requires BENDING_BUILD_CAD=ON")
endif()

include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)
configure_file(cmake/Version.h.in generated/bending/Version.h @ONLY)
add_subdirectory(src/core)
add_subdirectory(src/simulation)
add_subdirectory(src/application)

if(BENDING_BUILD_CAD)
    add_subdirectory(src/cad)
endif()
if(BENDING_BUILD_APP)
    add_subdirectory(src/project)
    add_subdirectory(src/viewer)
    add_subdirectory(src/fem)
    add_subdirectory(src/ui)
    add_subdirectory(src/app)
    include(cmake/Packaging.cmake)
endif()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
